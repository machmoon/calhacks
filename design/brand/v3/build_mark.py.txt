import math, sys
R=40.0; rc=7.0          # hex circumradius, corner radius
a=R*math.sqrt(3)/2      # inradius
g=12.0                  # gap between cells
D=2*a+g                 # centre spacing
A=(0.0,0.0); B=(-D/2, D*math.sqrt(3)/2); C=(D/2, D*math.sqrt(3)/2)
cx=(A[0]+B[0]+C[0])/3; cy=(A[1]+B[1]+C[1])/3
cells=[(x-cx,y-cy) for x,y in (A,B,C)]

def hexpath(c):
    x0,y0=c; pts=[]
    for k in range(6):
        t=math.radians(-90+60*k)
        pts.append((x0+R*math.cos(t), y0+R*math.sin(t)))
    # rounded polygon
    d=""
    for i in range(6):
        p0=pts[i-1]; p1=pts[i]; p2=pts[(i+1)%6]
        def towards(p,q,l):
            dx,dy=q[0]-p[0],q[1]-p[1]; n=math.hypot(dx,dy); return (p[0]+dx/n*l,p[1]+dy/n*l)
        s=towards(p1,p0,rc); e=towards(p1,p2,rc)
        d+=("M" if i==0 else "L")+f"{s[0]:.2f},{s[1]:.2f} Q{p1[0]:.2f},{p1[1]:.2f} {e[0]:.2f},{e[1]:.2f} "
    return d+"Z"

def neck(p,q,h0=9.0,hm=5.0):
    ux,uy=q[0]-p[0],q[1]-p[1]; n=math.hypot(ux,uy); ux/=n; uy/=n; vx,vy=-uy,ux
    s0=a-2.0; s1=n-a+2.0; sm=(s0+s1)/2; k=2*hm-h0
    P=lambda s,h:(p[0]+ux*s+vx*h, p[1]+uy*s+vy*h)
    a0=P(s0,h0); a1=P(s1,h0); c1=P(sm,k); b1=P(s1,-h0); b0=P(s0,-h0); c0=P(sm,-k)
    d=f"M{a0[0]:.2f},{a0[1]:.2f} Q{c1[0]:.2f},{c1[1]:.2f} {a1[0]:.2f},{a1[1]:.2f} L{b1[0]:.2f},{b1[1]:.2f} Q{c0[0]:.2f},{c0[1]:.2f} {b0[0]:.2f},{b0[1]:.2f} Z"
    return f'<path d="{d}"/>',""

def claude(c):   # Claude-style spark: 11 rounded rays of uneven length
    x0,y0=c; out=""
    lens=[19,15,18,14,19,16,18,14,19,15,17]
    for k,L in enumerate(lens):
        t=math.radians(-90+k*360/11+(3 if k%2 else -2))
        x1,y1=x0+math.cos(t)*3.5,y0+math.sin(t)*3.5
        x2,y2=x0+math.cos(t)*L,y0+math.sin(t)*L
        out+=f'<line x1="{x1:.2f}" y1="{y1:.2f}" x2="{x2:.2f}" y2="{y2:.2f}" stroke="black" stroke-width="3.6" stroke-linecap="round"/>'
    return out+f'<circle cx="{x0:.2f}" cy="{y0:.2f}" r="4.5"/>'

def creao(c):    # Creao-style mark: a stroked triangle and a ring sharing the triangle's right edge
    x0,y0=c
    tri=[(-19.0,12.0),(-6.5,-12.0),(4.5,12.0)]
    T=lambda p:f"{x0+p[0]:.2f},{y0+p[1]:.2f}"
    st='fill="none" stroke="black" stroke-width="4.2" stroke-linejoin="miter" stroke-miterlimit="8"'
    return (f'<path d="M{T(tri[0])} L{T(tri[1])} L{T(tri[2])} Z" {st}/>'
            f'<circle cx="{x0+10.5:.2f}" cy="{y0+2.0:.2f}" r="9.0" {st}/>')

def azure(c):    # two-blade A
    x0,y0=c
    L=[(-19,17),(-5,-18),(4,-18),(-9,17)]
    Rt=[(1,-9),(19,17),(-4,17),(8,9)]
    f=lambda P:'<polygon points="'+" ".join(f"{x0+x:.2f},{y0+y:.2f}" for x,y in P)+'" stroke="black" stroke-width="1.5" stroke-linejoin="round"/>'
    return f(L)+f(Rt)

pairs=[(0,1),(0,2),(1,2)]
rects=cuts=""
for i,j in pairs:
    r_,c_=neck(cells[i],cells[j]); rects+=r_; cuts+=c_
glyphs=creao(cells[0])+claude(cells[1])+azure(cells[2])
xs=[x for x,_ in cells]; ys=[y for _,y in cells]
pad=R+6
vb=(min(xs)-pad, min(ys)-pad, max(xs)-min(xs)+2*pad, max(ys)-min(ys)+2*pad)
side=max(vb[2],vb[3]); vx=vb[0]-(side-vb[2])/2; vy=vb[1]-(side-vb[3])/2
ink=sys.argv[1] if len(sys.argv)>1 else "#000"
svg=f'''<svg xmlns="http://www.w3.org/2000/svg" viewBox="{vx:.2f} {vy:.2f} {side:.2f} {side:.2f}" width="512" height="512">
<title>Wright</title>
<defs><mask id="m" maskUnits="userSpaceOnUse" x="{vx:.2f}" y="{vy:.2f}" width="{side:.2f}" height="{side:.2f}">
<g fill="white">{"".join(f'<path d="{hexpath(c)}"/>' for c in cells)}{rects}</g>
<g fill="black">{cuts}</g>
<g fill="black">{glyphs}</g>
</mask></defs>
<rect x="{vx:.2f}" y="{vy:.2f}" width="{side:.2f}" height="{side:.2f}" fill="{ink}" mask="url(#m)"/>
</svg>'''
print(svg)
